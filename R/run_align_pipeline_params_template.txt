# Pipeline config file for run_align_pipeline.R
#
# "keyword: value" lines (same rules as ground_returns.R's and align_dtms.R's own parameter
# files -- order within a section doesn't matter, blank lines and lines starting with "#" are
# ignored, a trailing "# comment" after a value is stripped, and only the first colon on a line
# separates the keyword from its value, so a value like "c:\..." or "EPSG:2856" is safe),
# grouped under section headers: [study_area], [defaults], [ref_set], [align_set], [ref_dtm],
# [align_dtm], [ref_dsm], [align_dsm], [align].
#
# [study_area] holds clip_polygon and target_crs, which are passed to every ground_returns.R and
# mosaic_dtm.R run -- set them only there. clip_polygon NOFILE (the default) uses every LAZ, DTM,
# and DSM tile in each folder, unmasked.
# [defaults] is optional and holds ground_returns.R parameters shared by both [ref_set] and
# [align_set]; a set can repeat a keyword itself to override the shared default just for it.
# [ref_set] and [align_set] are each one ground_returns.R run (see param_specs in
# ground_returns.R for the full list of recognized keywords) and must set input_dir and
# output_dir; each writes ground_count_ref_grid.flt there -- ground points per reference DTM
# cell -- which Align gets as its ground-return raster. [ref_dtm] and [align_dtm] each take
# their DTM from exactly one of:
#   tile_dir + output_dtm  -- a folder of tiles, mosaicked by mosaic_dtm.R (see param_specs in
#                             mosaic_dtm.R) and clipped to clip_polygon;
#   input_dtm + output_dtm -- one raster file (GeoTIFF, .flt, ...), run through mosaic_dtm.R the
#                             same way: clipped to clip_polygon, converted to meters, and
#                             reprojected to target_crs, then written to output_dtm;
#   existing_dtm alone     -- an already-mosaicked DTM used as-is (an ESRI GridFloat .flt, in
#                             meters and in target_crs; it is not clipped to clip_polygon).
# [ref_dsm] and [align_dsm] are optional and work the same way for digital surface models, with
# output_dsm/input_dsm/existing_dsm in place of output_dtm/input_dtm/existing_dtm; leave a
# section out (or set existing_dsm: NOFILE) for no DSM. [align] holds align_dtms.R /
# TWutils::align() parameters (see param_specs in align_dtms.R) -- do NOT set refDTM/alignDTM,
# refDSM/alignDSM, or refGrnd/alignGrnd there; they are always taken from the sections above.
#
# Every output is skipped if it already exists. After changing clip_polygon (or anything else),
# delete the outputs it affects so they're rebuilt.

[align]

# Maximum number of times to repeat the alignment calculation (in each of Align's global and
# tiled loops), and the RMSE change (m) below which Align stops iterating early; convergence 0
# runs every iteration. convergence needs alignSlope2 1.0.6 or later.
iterations: 10
convergence: 0.001

# Tukey-fence multiplier used to identify and exclude outliers (e.g. landslide sites) from the
# alignment calculation; k <= 0 filters more points as outliers.
k: 1.5

# Proportion of each iteration's calculated shift to actually apply (1.0 = apply it in full).
dampener: 1.0

# Output path (no extension) for the aligned DTM.
outDTM: c:\work\data\site1\elev_2023aligned

# Number of tiles in x (east-west) and y (north-south), and the fractional overlap between them.
# "auto" hands Align to align_supervisor.R, which picks the tiles from the study area's size
# and shape, then tries finer tilings while each lowers the ending RMSE by at least
# min_rmse_reduction, up to max_attempts Align runs in all, and keeps the best (see its
# header). Numbers run Align once with exactly those tiles.
tileNx: auto
tileNy: auto
overlap: 0.5

# Tile search (tileNx/tileNy: auto only; the values shown are the defaults):
# tile_size_m: 2000            # first attempt's target tile spacing (m)
# max_attempts: 3              # Align runs at most

# progress_interval_s: 30      # how often Align's progress is echoed
